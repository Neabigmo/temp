\section{Symmetrized-MGF dominance and main results}\label{sec:main-results}

\begin{definition}[Reflected defect]\label{def:reflected-defect}
Let $X$ be centered with $\E X^2=1$, and suppose $M_X$ is finite on $[-\rho,\rho]$. Define
\[
R_X(s):=\log M_X(s)+\log M_X(-s)-s^2,
\qquad
\Delta_\rho(X):=\sup_{|s|\le\rho}|R_X(s)|.
\]
The symmetrized-MGF dominance condition is the one-sided requirement $R_X(s)\ge0$ on the same interval. Under dominance, $\Delta_\rho(X)=\sup_{|s|\le\rho}R_X(s)$.
\end{definition}

\begin{definition}[Spherical Laplace discrepancy]\label{def:spherical-discrepancy}
For $r\ge1$, let $V$ be uniform on $S^{r-1}$ and set
\[
\Psi_r(z):=\E_V e^{zV_1}.
\]
For a nonnegative random variable $Q$ with the relevant exponential moments, define
\[
\mathfrak D_{r,\tau}(Q)
 :=\sup_{|t|\le\tau}
 \left|e^{-t^2/2}\E\Psi_r(t\sqrt Q)-1\right|.
\]
This is the Laplace observable obtained by averaging the radius-$\sqrt Q$ transform over a random direction; it is not a generic probability distance.
\end{definition}

\begin{proposition}[Naturality and separation of the dominance condition]\label{prop:naturality}
Let $X$ be centered with variance $\sigma^2$ and with an MGF in a neighborhood of the origin. If $X$ is infinitely divisible, then
\[
M_X(s)M_X(-s)\ge e^{\sigma^2s^2}
\]
for every $s$ such that both $M_X(s)$ and $M_X(-s)$ are finite. The converse is false, and symmetry alone does not imply this inequality.
\end{proposition}

\begin{proof}
For an infinitely divisible law with finite variance, the L\'evy--Khintchine representation gives, near the origin,
\[
\log M_X(s)+\log M_X(-s)-\sigma^2s^2
 =2\int\left(\cosh(sx)-1-\frac{s^2x^2}{2}\right)\nu(dx)\ge0,
\]
because $\cosh u-1\ge u^2/2$. For the converse, let $X$ take the values $3$ and $-1/3$ with probabilities $1/10$ and $9/10$, respectively. It is centered and has variance one. It is not infinitely divisible: if a bounded non-degenerate law of support diameter $d$ were an $n$-fold convolution of an i.i.d. root, that root would have support diameter at most $d/n$, and Popoviciu's inequality would give $\operatorname{Var}(X)/n\le d^2/(4n^2)$, which is impossible for large $n$. Its fourth cumulant is $46/9$, so analyticity at the origin gives
\[
R_X(s)=\frac{23}{54}s^4+O(s^6).
\]
Consequently $R_X(s)\ge0$ on some nontrivial neighborhood of the origin, and this example is asymmetric. Finally, for a Rademacher variable $\varepsilon$, $R_\varepsilon(s)=2\log\cosh s-s^2=-s^4/6+O(s^6)$, which is negative for all sufficiently small nonzero $s$.
\end{proof}

\begin{theorem}[Exact PSD rigidity]\label{thm:exact-psd}
Let $X=(X_1,\ldots,X_d)$ have independent coordinates with $\E X_i=0$ and $\operatorname{Var}(X_i)=\sigma_i^2$. Let $A=A^{\mathsf T}\succeq0$ have rank $r\ge1$, and suppose
\[
X^{\mathsf T}AX\sim\chi_r^2.
\]
For every active coordinate $i$ with $A_{ii}>0$, assume that $M_i(s)=\E e^{sX_i}$ is finite in a neighborhood of the origin and
\[
M_i(s)M_i(-s)\ge e^{\sigma_i^2s^2}
\]
there. Then every active $X_i$ is Gaussian, $X_i\sim\Normal(0,\sigma_i^2)$.
\end{theorem}

\begin{corollary}[Exact sample-variance characterization under dominance]\label{cor:exact-sample-var}
Let $n\ge3$ and let $X_1,\ldots,X_n$ be i.i.d., centered, and variance one. If $M_X$ is finite near the origin,
\[
M_X(s)M_X(-s)\ge e^{s^2}
\]
there, and
\[
\sum_{i=1}^n(X_i-\bar X)^2\sim\chi^2_{n-1},
\]
then $X_i\sim\Normal(0,1)$.
\end{corollary}

\begin{theorem}[Sharp reflection stability]\label{thm:sharp-upper}
Fix $\rho,\tau,K>0$. There exist constants $C<\infty$ and $\Delta_0\in(0,e^{-e})$, depending only on $(\rho,\tau,K)$, such that the following holds. If
\[
\E X=0,\qquad \E X^2=1,\qquad \E e^{2\tau|X|}\le K,
\]
and
\[
|R_X(s)|\le\Delta,
\qquad |s|\le\rho,
\]
for some $0<\Delta\le\Delta_0$, then
\[
\dK(X,\Normal(0,1))
\le C\sqrt{\frac{\log\log(1/\Delta)}{\log(1/\Delta)}}.
\]
\end{theorem}

\begin{theorem}[Sample-variance stability]\label{thm:sample-var}
Fix $n\ge3$, $\tau,K>0$, and $0<\rho'<\tau\sqrt{(n-1)/n}$. There are constants $C<\infty$ and $\varepsilon_0>0$, depending only on $(n,\tau,K,\rho')$, such that the following holds. Let $X_1,\ldots,X_n$ be i.i.d. copies of a centered variance-one $X$ with $\E e^{2\tau|X|}\le K$, and assume $R_X(s)\ge0$ for $|s|\le\tau$. If
\[
Q_n=\sum_{i=1}^n(X_i-\bar X)^2,
\qquad
\mathfrak D_{n-1,\tau}(Q_n)\le\varepsilon\le\varepsilon_0,
\]
then
\[
\dK(X,\Normal(0,1))
\le C\sqrt{\frac{\log\log(1/\varepsilon)}{\log(1/\varepsilon)}}.
\]
\end{theorem}

\begin{theorem}[Laguerre lower bound]\label{thm:sharpness}
There are constants $\rho,\tau,K,c,C>0$ and centered variance-one random variables $W_m$ such that the same uniform exponential-integrability and local dominance assumptions hold, $\Delta_m:=\Delta_\rho(W_m)\downarrow0$, and
\[
\log\frac1{\Delta_m}=2m\log m+O(m),
\qquad
\dK(W_m,\Normal(0,1))\ge c m^{-1/2}.
\]
Consequently
\[
\dK(W_m,\Normal(0,1))
\ge c'\sqrt{\frac{\log\log(1/\Delta_m)}{\log(1/\Delta_m)}}
\]
for all sufficiently large $m$.
\end{theorem}

\begin{proposition}[Strong-distance obstruction]\label{prop:tv}
There is no uniform total-variation stability theorem under the assumptions of \cref{thm:sharp-upper}: a standardized Poisson lattice family can satisfy a fixed exponential-integrability bound and have reflected defect tending to zero while its total-variation distance from every absolutely continuous Gaussian law equals one.
\end{proposition}
